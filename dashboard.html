Supabase SQL Schema & Full RLS Setup

ຄັດລອກ SQL ນີ້ໄປວາງໃສ່ Supabase SQL Editor ເພື່ອສ້າງ Tables ແລະ RLS (Full CRUD): 
-- ========================================================
-- 1. CREATE TABLES
-- ========================================================

CREATE TABLE IF NOT EXISTS admin (
    admin_id VARCHAR PRIMARY KEY,
    password VARCHAR NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now())
);

INSERT INTO admin (admin_id, password) 
VALUES ('admin', '123456')
ON CONFLICT (admin_id) DO NOTHING;

CREATE TABLE IF NOT EXISTS users (
    id_user VARCHAR PRIMARY KEY,
    password VARCHAR NOT NULL,
    department VARCHAR,
    create_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now())
);

CREATE TABLE IF NOT EXISTS evaluatees (
    id_evaluatee VARCHAR PRIMARY KEY,
    name VARCHAR NOT NULL,
    surname VARCHAR,
    position VARCHAR,
    department VARCHAR,
    create_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now())
);

CREATE TABLE IF NOT EXISTS lists (
    id_list VARCHAR PRIMARY KEY,
    name VARCHAR NOT NULL,
    create_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now())
);

CREATE TABLE IF NOT EXISTS datas (
    id_data VARCHAR PRIMARY KEY,
    id_user VARCHAR REFERENCES users(id_user) ON DELETE CASCADE ON UPDATE CASCADE,
    id_evaluatee VARCHAR REFERENCES evaluatees(id_evaluatee) ON DELETE CASCADE ON UPDATE CASCADE,
    time_add INT,
    model_add VARCHAR,
    score NUMERIC,
    comment TEXT,
    create_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now())
);

-- ========================================================
-- 2. ENABLE ROW LEVEL SECURITY (RLS) & FULL CRUD POLICIES
-- ========================================================

ALTER TABLE admin ENABLE ROW LEVEL SECURITY;
ALTER TABLE users ENABLE ROW LEVEL SECURITY;
ALTER TABLE evaluatees ENABLE ROW LEVEL SECURITY;
ALTER TABLE lists ENABLE ROW LEVEL SECURITY;
ALTER TABLE datas ENABLE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS "Enable full access for admin" ON admin;
DROP POLICY IF EXISTS "Enable full access for users" ON users;
DROP POLICY IF EXISTS "Enable full access for evaluatees" ON evaluatees;
DROP POLICY IF EXISTS "Enable full access for lists" ON lists;
DROP POLICY IF EXISTS "Enable full access for datas" ON datas;

CREATE POLICY "Enable full access for admin" ON admin FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Enable full access for users" ON users FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Enable full access for evaluatees" ON evaluatees FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Enable full access for lists" ON lists FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Enable full access for datas" ON datas FOR ALL USING (true) WITH CHECK (true);
